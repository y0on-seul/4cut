    <script>
        const video = document.getElementById('webcam');
        const countdownOverlay = document.getElementById('countdown-overlay');
        const flashEffect = document.getElementById('flash-effect');
        const startBtn = document.getElementById('start-btn');
        const captureSection = document.getElementById('capture-section');
        const resultSection = document.getElementById('result-section');
        const frame = document.getElementById('four-cut-frame');
        const finalOutputImage = document.getElementById('final-output-image');

        let stream = null;
        let capturedPhotos = [];
        let currentStickerUrl = null;

        // 오늘 날짜 설정 (YYYY. MM. DD)
        function setFrameDate() {
            const today = new Date();
            const year = today.getFullYear();
            const month = String(today.getMonth() + 1).padStart(2, '0');
            const day = String(today.getDate()).padStart(2, '0');
            document.getElementById('frame-date').innerText = `${year}. ${month}. ${day}`;
        }

        // 웹캠 에러 핸들링 및 시작
        async function initCamera() {
            try {
                stream = await navigator.mediaDevices.getUserMedia({
                    video: { 
                        aspectRatio: 3/4,
                        facingMode: 'user'
                    }, 
                    audio: false 
                });
                video.srcObject = stream;
            } catch (err) {
                alert('카메라 권한을 허용해 주셔야 앱 이용이 가능합니다.');
                console.error("Camera init error:", err);
            }
        }

        // 촬영 프로세스 시작
        async function startShooting() {
            startBtn.style.display = 'none'; // 셔터 버튼 숨김
            capturedPhotos = [];
            
            for (let i = 0; i < 4; i++) {
                await runCountdown(3);
                triggerFlash();
                captureCanvasToMemory(i);
                await new Promise(resolve => setTimeout(resolve, 500)); // 플래시 및 정리 대기
            }

            endShooting();
        }

        // 카운트다운 로직
        function runCountdown(seconds) {
            return new Promise((resolve) => {
                let counter = seconds;
                countdownOverlay.innerText = counter;
                countdownOverlay.classList.add('active');

                const interval = setInterval(() => {
                    counter--;
                    if (counter > 0) {
                        countdownOverlay.innerText = counter;
                    } else {
                        clearInterval(interval);
                        countdownOverlay.classList.remove('active');
                        resolve();
                    }
                }, 1000);
            });
        }

        // 플래시 애니메이션 효과
        function triggerFlash() {
            flashEffect.style.animation = 'none';
            flashEffect.offsetHeight; // 트리거 리플로우
            flashEffect.style.animation = 'flash-animation 0.4s ease-out';
        }

        // 현재 웹캠 화면을 캡처하여 메모리에 임시 슬롯 저장
        function captureCanvasToMemory(index) {
            const canvas = document.createElement('canvas');
            canvas.width = video.videoWidth || 640;
            canvas.height = video.videoHeight || 480;
            const ctx = canvas.getContext('2d');
            
            // 거울 모드 비디오 캡처를 위해 canvas 좌우 반전 후 드로잉
            ctx.translate(canvas.width, 0);
            ctx.scale(-1, 1);
            ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
            
            const dataUrl = canvas.toDataURL('image/jpeg');
            capturedPhotos.push(dataUrl);

            // 해당 슬롯에 미리 이미지 주입
            const slot = document.getElementById(`slot-${index}`);
            slot.innerHTML = `<img src="${dataUrl}" alt="cut-${index}">`;
        }

        // 촬영 완료 후 결과창 렌더링
        function endShooting() {
            setFrameDate();
            captureSection.style.display = 'none';
            resultSection.style.display = 'flex';
            
            // 스트림 정지하여 카메라 끔
            if (stream) {
                stream.getTracks().forEach(track => track.stop());
            }

            updateFinalImage(); // 초기 완성본 이미지화
        }

        // 프레임 배경 및 텍스트 컬러 변경 기능
        function changeFrameColor(bgColor, textColor, element) {
            frame.style.setProperty('--frame-color', bgColor);
            frame.style.setProperty('--frame-text-color', textColor);
            
            // Active 클래스 교체
            document.querySelectorAll('.color-opt').forEach(opt => opt.classList.remove('active'));
            element.classList.add('active');

            updateFinalImage();
        }

        // 커스텀 스티커 업로드 처리
        function handleStickerUpload(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                currentStickerUrl = e.target.result;
                applyStickers();
            }
            reader.readAsDataURL(file);
        }

        // 1번 슬롯 좌측 상단, 4번 슬롯 우측 하단에 스티커 합성
        function applyStickers() {
            if (!currentStickerUrl) return;

            // 기존 스티커 제거
            document.querySelectorAll('.sticker').forEach(el => el.remove());

            const slot1 = document.getElementById('slot-0');
            const slot4 = document.getElementById('slot-3');

            const sticker1 = document.createElement('img');
            sticker1.src = currentStickerUrl;
            sticker1.className = 'sticker sticker-tl';

            const sticker4 = document.createElement('img');
            sticker4.src = currentStickerUrl;
            sticker4.className = 'sticker sticker-br';

            slot1.appendChild(sticker1);
            slot4.appendChild(sticker4);

            // 스티커 이미지가 로드된 후 최종 캡처 업데이트
            sticker4.onload = () => {
                updateFinalImage();
            };
        }

        // html2canvas를 활용해 46 사이즈 인화(기본 2:3 비율 프레임)에 최적화된 고해상도 이미지 파일 생성
        function updateFinalImage() {
            // 원본 DOM을 표시한 상태에서 캡처 진행
            frame.style.display = 'flex';
            finalOutputImage.style.display = 'none';

            // 고해상도 인화를 위한 스케일업 배율 설정 (46사이즈 맞춤 해상도 약 1200x1800 확보)
            html2canvas(frame, {
                scale: 3.5, 
                useCORS: true,
                logging: false,
                backgroundColor: null
            }).then(canvas => {
                const finalDataUrl = canvas.toDataURL('image/png');
                finalOutputImage.src = finalDataUrl;
                
                // 모바일 유저가 길게 눌러 다운로드하기 쉽도록 실제 렌더링은 img 태그만 노출
                frame.style.display = 'none';
                finalOutputImage.style.styleFloat = 'none';
                finalOutputImage.style.display = 'block';
            });
        }

        // 초기 상태로 리셋
        function resetApp() {
            // 기존 스티커 데이터 비우기
            currentStickerUrl = null;
            document.getElementById('sticker-input').value = "";
            document.querySelectorAll('.sticker').forEach(el => el.remove());
            
            // UI 토글
            resultSection.style.display = 'none';
            captureSection.style.display = 'flex';
            startBtn.style.display = 'flex';

            // 기본 화이트 프레임 원복
            changeFrameColor('#ffffff', '#000000', document.querySelector('.color-opt'));

            // 카메라 재구동
            initCamera();
        }

        // 앱 시작 시 카메라 최초 로드
        window.onload = initCamera;
    </script>
